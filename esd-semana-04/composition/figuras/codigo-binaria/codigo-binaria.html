def buscar_binaria(a, clave):
    bajo, alto = 0, len(a) - 1
    while bajo <= alto:
        centro = (bajo + alto) // 2
        if a[centro] == clave:
            return centro
        if clave < a[centro]:
            alto = centro - 1
        else:
            bajo = centro + 1
    return -1

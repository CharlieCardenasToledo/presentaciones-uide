def setUp(self):
    self.pila = PilaArreglo(3)
 
def test_desapilar_devuelve_en_orden_lifo(self):
    for x in (10, 20, 30):
        self.pila.apilar(x)
        self.assertEqual(self.pila.desapilar(), 30)
        self.assertEqual(self.pila.desapilar(), 20)
 
def test_desapilar_pila_vacia_falla(self):
    with self.assertRaises(IndexError):
        self.pila.desapilar()
